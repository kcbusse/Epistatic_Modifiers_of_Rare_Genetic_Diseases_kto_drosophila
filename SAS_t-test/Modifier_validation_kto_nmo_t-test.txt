#T-test comparing kto genotype and SNP genotype including controls#

#FEMALE T-TEST#
proc univariate data=work.nmo_F normal;
    class ktogenotype;
    var 'Motility(s)'n;
run;

ods output TTests=work.ttest_NmoF;

proc ttest data=work.nmo_F sides=2 plots(showh0);
    class ktogenotype;
    var 'Motility(s)'n;
run;

data work.ttest_NmoF_scientific;
    set work.ttest_NmoF;
    Probt_scientific = put(Probt, BEST32.-L);
Run;
