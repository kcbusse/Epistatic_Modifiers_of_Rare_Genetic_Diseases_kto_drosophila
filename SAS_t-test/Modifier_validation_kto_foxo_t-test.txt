#T-test comparing kto genotype and SNP genotype including controls#

#FEMALE T-TEST#
proc univariate data=work.foxo_F normal;
    class ktogenotype;
    var 'Motility(s)'n;
run;

ods output TTests=work.ttest_FoxoF;

proc ttest data=work.foxo_F sides=2 plots(showh0);
    class ktogenotype;
    var 'Motility(s)'n;
run;

data work.ttest_FoxoF_scientific;
    set work.ttest_FoxoF;
    Probt_scientific = put(Probt, BEST32.-L);
Run;
