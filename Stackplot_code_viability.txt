# -----------------------------
# Head Involution and Dorsal Issues Morphology Data 
# -----------------------------

##input your sheet data 
dorsal_head <- matrix(c(3, 28, 27,
                        426, 409, 285),
                      nrow = 2, byrow = TRUE)

rownames(dorsal_head) <- c("Dorsal/Head defect", "WT")
colnames(dorsal_head) <- c("Control", "kto20228C", "kto20228D")

prop_dorshead <- prop.table(dorsal_head, margin = 2)

#Run fisher's exact test for each null against control
p1 <- fisher.test(dorsal_head[, c("Control", "kto20228C")])$p.value
p2 <- fisher.test(dorsal_head[, c("Control", "kto20228D")])$p.value

get_stars <- function(p) {
  if (p < 0.001) "***"
  else if (p < 0.01) "**"
  else if (p < 0.05) "*"
  else "ns"
}

s1 <- get_stars(p1)
s2 <- get_stars(p2)


#format your X-axis for your future bargraph
x_labels <- c(
  expression(italic(kto)^{"+ / +"}),
  expression(GFP - italic(kto)^{20228*C}),
  expression(GFP - italic(kto)^{20228*D})
)

#Plot the stacked bargraph 
par(mar = c(5, 4, 4, 8), xpd = TRUE)

bp <- barplot(prop_dorshead,
              col = c("black", "white"),
              ylim = c(0, 1.2),
              ylab = "Proportion of Embryos",
              xlab = "Genotype",
              names.arg = x_labels)

#Add the legend in
legend("topright",
inset = c(-0.28, 0), #change this if legend position needs to alter
legend = c("Dorsal/Head\nDefect", "WT"), #wraps the text on long title to avoid cutting it off
fill = c("black", "white"),
bty = "n",
xpd = TRUE)

#Use this function if you need to shift the postion of the significance astrisks
#shift1 <- 0.6
#shift2 <- 1.21

#Add the significance values to the stack plot 
text(x = mean(bp[c(1,2)]),
     y = 1.09,
     labels = s1,
     cex = 1.2)

text(x = mean(bp[c(1,3)]),
     y = 1.15,
     labels = s2,
     cex = 1.2)
